cmake_minimum_required(VERSION 3.22)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(ttcp LANGUAGES C)
  set(CMAKE_C_STANDARD 11)
  set(CMAKE_C_STANDARD_REQUIRED ON)
endif()

include(CheckFunctionExists)
include(CheckIncludeFiles)
include(CheckSymbolExists)
include(CheckTypeSize)

set(TTCP_COMPILE_DEFINITIONS)
if(NOT WIN32)
  if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    list(APPEND TTCP_COMPILE_DEFINITIONS _POSIX_C_SOURCE=200112L)
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    list(APPEND TTCP_COMPILE_DEFINITIONS _DEFAULT_SOURCE=1)
  endif()
endif()
set(TTCP_REQUIRED_DEFINITIONS ${TTCP_COMPILE_DEFINITIONS})
list(TRANSFORM TTCP_REQUIRED_DEFINITIONS PREPEND -D)
set(CMAKE_REQUIRED_DEFINITIONS ${TTCP_REQUIRED_DEFINITIONS})

set(TTCP_LIBRARIES)
if(WIN32)
  list(APPEND TTCP_LIBRARIES ws2_32)
else()
  check_function_exists(socket TTCP_SOCKET_IN_LIBC)
  if(NOT TTCP_SOCKET_IN_LIBC)
    find_library(TTCP_SOCKET_LIBRARY socket)
    if(TTCP_SOCKET_LIBRARY)
      list(APPEND TTCP_LIBRARIES ${TTCP_SOCKET_LIBRARY})
    endif()
  endif()

  set(CMAKE_REQUIRED_LIBRARIES ${TTCP_LIBRARIES})
  check_function_exists(gethostbyaddr TTCP_GETHOSTBYADDR_IN_LIBC)
  if(NOT TTCP_GETHOSTBYADDR_IN_LIBC)
    find_library(TTCP_NSL_LIBRARY nsl)
    if(TTCP_NSL_LIBRARY)
      list(APPEND TTCP_LIBRARIES ${TTCP_NSL_LIBRARY})
    endif()
  endif()

  set(CMAKE_REQUIRED_LIBRARIES ${TTCP_LIBRARIES})
  check_function_exists(socket TTCP_HAVE_SOCKET)
  if(NOT TTCP_HAVE_SOCKET)
    find_library(TTCP_NETWORK_LIBRARY network)
    if(TTCP_NETWORK_LIBRARY)
      list(APPEND TTCP_LIBRARIES ${TTCP_NETWORK_LIBRARY})
    endif()
  endif()
endif()
set(CMAKE_REQUIRED_LIBRARIES ${TTCP_LIBRARIES})

check_include_files("winsock2.h;ws2tcpip.h" HAVE_WINSOCK2_H)
check_include_files(windows.h HAVE_WINDOWS_H)
check_include_files(sys/types.h HAVE_SYS_TYPES_H)
check_include_files("sys/types.h;sys/socket.h" HAVE_SYS_SOCKET_H)
check_include_files("sys/types.h;sys/socket.h;arpa/inet.h" HAVE_ARPA_INET_H)
check_include_files("sys/types.h;netinet/in.h" HAVE_NETINET_IN_H)
check_include_files(netdb.h HAVE_NETDB_H)
check_include_files(unistd.h HAVE_UNISTD_H)
check_include_files(sys/time.h HAVE_SYS_TIME_H)
check_include_files(sys/resource.h HAVE_SYS_RESOURCE_H)
check_include_files("sys/types.h;sys/select.h" HAVE_SYS_SELECT_H)

set(CMAKE_EXTRA_INCLUDE_FILES)
if(HAVE_SYS_TYPES_H)
  list(APPEND CMAKE_EXTRA_INCLUDE_FILES sys/types.h)
endif()
if(HAVE_SYS_SOCKET_H)
  list(APPEND CMAKE_EXTRA_INCLUDE_FILES sys/socket.h)
elseif(HAVE_WINSOCK2_H)
  list(APPEND CMAKE_EXTRA_INCLUDE_FILES winsock2.h)
endif()
check_type_size(socklen_t TTCP_SOCKLEN_T)
if(TTCP_SOCKLEN_T)
  set(HAVE_SOCKLEN_T 1)
endif()
unset(CMAKE_EXTRA_INCLUDE_FILES)

if(HAVE_WINSOCK2_H)
  set(TTCP_NETWORK_HEADERS "winsock2.h;ws2tcpip.h")
else()
  set(TTCP_NETWORK_HEADERS netdb.h)
endif()
check_symbol_exists(getaddrinfo "${TTCP_NETWORK_HEADERS}" HAVE_GETADDRINFO)
check_symbol_exists(gethostbyname "${TTCP_NETWORK_HEADERS}" HAVE_GETHOSTBYNAME)
check_symbol_exists(gettimeofday sys/time.h HAVE_GETTIMEOFDAY)
check_symbol_exists(getrusage sys/resource.h HAVE_GETRUSAGE)
if(HAVE_SYS_SELECT_H)
  check_symbol_exists(select "sys/types.h;sys/select.h" HAVE_SELECT)
elseif(HAVE_WINSOCK2_H)
  check_symbol_exists(select winsock2.h HAVE_SELECT)
endif()

configure_file(ttcp_config.h.in ttcp_config.h)

set(TTCP_SOURCE ttcp.c)
if(COMMAND brlcad_addexec)
  brlcad_addexec(ttcp ${TTCP_SOURCE} ${TTCP_LIBRARIES} FOLDER Util)
else()
  add_executable(ttcp ${TTCP_SOURCE})
  if(TTCP_LIBRARIES)
    target_link_libraries(ttcp PRIVATE ${TTCP_LIBRARIES})
  endif()
  install(TARGETS ttcp RUNTIME DESTINATION bin)
endif()

target_include_directories(ttcp PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_definitions(ttcp PRIVATE ${TTCP_COMPILE_DEFINITIONS})
